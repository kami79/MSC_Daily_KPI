SELECT
  S.NE_NAME,
  S.DATE_ID,
  S.HOUR_ID,

  -- 1. Current Call Setup Rate (per second)
  CAST(ROUND(
    CASE 
      WHEN S.NE_VERSION LIKE 'R4.%' OR S.NE_VERSION LIKE 'R5.%' THEN -1.0
      ELSE APP.pmAvMedStreamChanSetUpRate_avg / 2.005
    END, 1) AS NUMERIC(10,1)) AS [Current Call Setup Rate (per second)],

  -- 2. Data Availability (%)
  CAST(ROUND(
    CASE 
      WHEN ((100.0 * DA.DATACOVERAGE) / NULLIF(DA.PERIOD_DURATION, 0)) > 100.0 THEN 100.0
      ELSE (100.0 * DA.DATACOVERAGE) / NULLIF(DA.PERIOD_DURATION, 0)
    END, 1) AS NUMERIC(10,1)) AS [Data Availability (%)],

  -- 3. Incoming User Plane Success Rate (%)
  CAST(ROUND(
    CASE
      WHEN ((((COALESCE(R4.pmRtpReceivedPkts,0) + COALESCE(R5.pmRtpReceivedPktsHi,0)*2147483648.0 + COALESCE(UR.pmRtpReceivedPktsHi,0)*2147483648.0 +
               COALESCE(R5.pmRtpReceivedPktsLo,0) + COALESCE(UR.pmRtpReceivedPktsLo,0) +
               COALESCE(R4.pmRtpLostPkts,0) + COALESCE(R5.pmRtpLostPkts,0) + COALESCE(UR.pmRtpLostPkts,0))
              -
              (COALESCE(R4.pmRtpDiscardedPkts,0) + COALESCE(R4.pmRtpLostPkts,0) +
               COALESCE(R5.pmRtpDiscardedPkts,0) + COALESCE(R5.pmRtpLostPkts,0) +
               COALESCE(UR.pmRtpDiscardedPkts,0) + COALESCE(UR.pmRtpLostPkts,0))) * 100.0)
             / NULLIF(
               COALESCE(R4.pmRtpReceivedPkts,0) + COALESCE(R5.pmRtpReceivedPktsHi,0)*2147483648.0 + COALESCE(UR.pmRtpReceivedPktsHi,0)*2147483648.0 +
               COALESCE(R5.pmRtpReceivedPktsLo,0) + COALESCE(UR.pmRtpReceivedPktsLo,0) +
               COALESCE(R4.pmRtpLostPkts,0) + COALESCE(R5.pmRtpLostPkts,0) + COALESCE(UR.pmRtpLostPkts,0), 0)) > 100.0
      THEN 100.0
      ELSE
        (((COALESCE(R4.pmRtpReceivedPkts,0) + COALESCE(R5.pmRtpReceivedPktsHi,0)*2147483648.0 + COALESCE(UR.pmRtpReceivedPktsHi,0)*2147483648.0 +
           COALESCE(R5.pmRtpReceivedPktsLo,0) + COALESCE(UR.pmRtpReceivedPktsLo,0) +
           COALESCE(R4.pmRtpLostPkts,0) + COALESCE(R5.pmRtpLostPkts,0) + COALESCE(UR.pmRtpLostPkts,0))
          -
          (COALESCE(R4.pmRtpDiscardedPkts,0) + COALESCE(R4.pmRtpLostPkts,0) +
           COALESCE(R5.pmRtpDiscardedPkts,0) + COALESCE(R5.pmRtpLostPkts,0) +
           COALESCE(UR.pmRtpDiscardedPkts,0) + COALESCE(UR.pmRtpLostPkts,0))) * 100.0)
         / NULLIF(
           COALESCE(R4.pmRtpReceivedPkts,0) + COALESCE(R5.pmRtpReceivedPktsHi,0)*2147483648.0 + COALESCE(UR.pmRtpReceivedPktsHi,0)*2147483648.0 +
           COALESCE(R5.pmRtpReceivedPktsLo,0) + COALESCE(UR.pmRtpReceivedPktsLo,0) +
           COALESCE(R4.pmRtpLostPkts,0) + COALESCE(R5.pmRtpLostPkts,0) + COALESCE(UR.pmRtpLostPkts,0), 0)
    END, 1) AS NUMERIC(10,1)) AS [Incoming User Plane Success Rate (%)],

  -- 4. IP CS CN Transport QoS, Jitter (%)
  CAST(ROUND(
    CASE 
      WHEN S.NE_VERSION LIKE 'R4.%' OR S.NE_VERSION LIKE 'R5.0%' THEN
        ((COALESCE(J.pmIpCnConnMeasuredJitter0,0) + COALESCE(J.pmIpCnConnMeasuredJitter1,0) +
          COALESCE(J.pmIpCnConnMeasuredJitter2,0) + COALESCE(J.pmIpCnConnMeasuredJitter3,0) +
          COALESCE(J.pmIpCnConnMeasuredJitter4,0)) * 100.0)
         / NULLIF(
          COALESCE(J.pmIpCnConnMeasuredJitter0,0) + COALESCE(J.pmIpCnConnMeasuredJitter1,0) +
          COALESCE(J.pmIpCnConnMeasuredJitter2,0) + COALESCE(J.pmIpCnConnMeasuredJitter3,0) +
          COALESCE(J.pmIpCnConnMeasuredJitter4,0) + COALESCE(J.pmIpCnConnMeasuredJitter5,0), 0)
      ELSE
        ((COALESCE(J.pmIpCnConnMeasuredJitter0,0) + COALESCE(J.pmIpCnConnMeasuredJitter1,0) +
          COALESCE(J.pmIpCnConnMeasuredJitter2,0) + COALESCE(J.pmIpCnConnMeasuredJitter3,0)) * 100.0)
         / NULLIF(
          COALESCE(J.pmIpCnConnMeasuredJitter0,0) + COALESCE(J.pmIpCnConnMeasuredJitter1,0) +
          COALESCE(J.pmIpCnConnMeasuredJitter2,0) + COALESCE(J.pmIpCnConnMeasuredJitter3,0) +
          COALESCE(J.pmIpCnConnMeasuredJitter4,0) + COALESCE(J.pmIpCnConnMeasuredJitter5,0), 0)
    END, 1) AS NUMERIC(10,1)) AS [IP CS CN Transport QoS, Jitter (%)],

  -- 5. IP signalling quality (as %)
  CAST(ROUND(
    CASE
      WHEN ((((COALESCE(IP.pmIpInReceives,0) + COALESCE(IP.pmIpOutRequests,0))
              - (COALESCE(IP.pmIpInDiscards,0) + COALESCE(IP.pmIpOutDiscards,0) + COALESCE(IP.pmIpInAddrErrors,0) + COALESCE(IP.pmIpInHdrErrors,0) + COALESCE(IP.pmIpInUnknownProtos,0))) * 100.0)
             / NULLIF(COALESCE(IP.pmIpInReceives,0) + COALESCE(IP.pmIpOutRequests,0), 0)) > 100.0
      THEN 100.0
      ELSE
        (((COALESCE(IP.pmIpInReceives,0) + COALESCE(IP.pmIpOutRequests,0))
          - (COALESCE(IP.pmIpInDiscards,0) + COALESCE(IP.pmIpOutDiscards,0) + COALESCE(IP.pmIpInAddrErrors,0) + COALESCE(IP.pmIpInHdrErrors,0) + COALESCE(IP.pmIpInUnknownProtos,0))) * 100.0)
         / NULLIF(COALESCE(IP.pmIpInReceives,0) + COALESCE(IP.pmIpOutRequests,0), 0)
    END, 1) AS NUMERIC(10,1)) AS [IP signalling quality (as %)],

  -- 6. IP VoIP Transport QoS, Jitter (%)
  CAST(ROUND(
    CASE 
      WHEN S.NE_VERSION LIKE 'R4.%' THEN -1.0
      ELSE
        ((COALESCE(J.pmVoIpConnMeasuredJitter0,0) + COALESCE(J.pmVoIpConnMeasuredJitter1,0) +
          COALESCE(J.pmVoIpConnMeasuredJitter2,0) + COALESCE(J.pmVoIpConnMeasuredJitter3,0)) * 100.0)
         / NULLIF(
          COALESCE(J.pmVoIpConnMeasuredJitter0,0) + COALESCE(J.pmVoIpConnMeasuredJitter1,0) +
          COALESCE(J.pmVoIpConnMeasuredJitter2,0) + COALESCE(J.pmVoIpConnMeasuredJitter3,0) +
          COALESCE(J.pmVoIpConnMeasuredJitter4,0) + COALESCE(J.pmVoIpConnMeasuredJitter5,0) +
          COALESCE(J.pmVoIpConnMeasuredJitter6,0) + COALESCE(J.pmVoIpConnMeasuredJitter7,0) +
          COALESCE(J.pmVoIpConnMeasuredJitter8,0), 0)
    END, 1) AS NUMERIC(10,1)) AS [IP VoIP Transport QoS, Jitter (%)],

  -- 7. Iu over IP Transport QoS, Jitter (%)
  CAST(ROUND(
    CASE 
      WHEN S.NE_VERSION LIKE 'R4.%' OR S.NE_VERSION LIKE 'R5.0%' THEN -1.0
      ELSE
        ((COALESCE(J.pmIpRanConnMeasuredJitter0,0) + COALESCE(J.pmIpRanConnMeasuredJitter1,0) +
          COALESCE(J.pmIpRanConnMeasuredJitter2,0) + COALESCE(J.pmIpRanConnMeasuredJitter3,0)) * 100.0)
         / NULLIF(
          COALESCE(J.pmIpRanConnMeasuredJitter0,0) + COALESCE(J.pmIpRanConnMeasuredJitter1,0) +
          COALESCE(J.pmIpRanConnMeasuredJitter2,0) + COALESCE(J.pmIpRanConnMeasuredJitter3,0) +
          COALESCE(J.pmIpRanConnMeasuredJitter4,0) + COALESCE(J.pmIpRanConnMeasuredJitter5,0), 0)
    END, 1) AS NUMERIC(10,1)) AS [Iu over IP Transport QoS, Jitter (%)],

  -- 8. Narrowband Signalling Quality over TDM (as %)
  CAST(ROUND(
    CASE
      WHEN ((((COALESCE(MTP.pmNoOfMSUReceived,0) + COALESCE(MTP.pmNoOfMSUTransmitted,0))
              - (COALESCE(MTP.pmNoOfNacks,0) + COALESCE(MTP.pmNoOfSuReceivedInError,0))) * 100.0)
             / NULLIF(COALESCE(MTP.pmNoOfMSUReceived,0) + COALESCE(MTP.pmNoOfMSUTransmitted,0), 0)) > 100.0
      THEN 100.0
      ELSE
        (((COALESCE(MTP.pmNoOfMSUReceived,0) + COALESCE(MTP.pmNoOfMSUTransmitted,0))
          - (COALESCE(MTP.pmNoOfNacks,0) + COALESCE(MTP.pmNoOfSuReceivedInError,0))) * 100.0)
         / NULLIF(COALESCE(MTP.pmNoOfMSUReceived,0) + COALESCE(MTP.pmNoOfMSUTransmitted,0), 0)
    END, 1) AS NUMERIC(10,1)) AS [Narrowband Signalling Quality over TDM (as %)],

  -- 9. Ratio of Late Packet for External IP Interface connections (%)
  CAST(ROUND(
    (COALESCE(J.pmLatePktsExt,0) * 100.0)
    / NULLIF(COALESCE(J.pmLatePktsExt,0) + (COALESCE(J.pmSuccTransmittedPktsExtHi,0) * 2147483648.0) + COALESCE(J.pmSuccTransmittedPktsExtLo,0), 0),
    1) AS NUMERIC(10,1)) AS [Ratio of Late Packet for External IP Interface connections (%)],

  -- 10. Ratio of Late Packet for Iu over IP connections (%)
  CAST(ROUND(
    (COALESCE(J.pmLatePktsIpRan,0) * 100.0)
    / NULLIF(COALESCE(J.pmLatePktsIpRan,0) + COALESCE(J.pmSuccTransmittedPktsIpRan,0), 0),
    1) AS NUMERIC(10,1)) AS [Ratio of Late Packet for Iu over IP connections (%)],

  -- 11. Ratio of Late packets for IP CS CN (%)
  CAST(ROUND(
    (COALESCE(J.pmLatePktsIpCn,0) * 100.0)
    / NULLIF(COALESCE(J.pmLatePktsIpCn,0) + COALESCE(J.pmSuccTransmittedPktsIpCn,0), 0),
    1) AS NUMERIC(10,1)) AS [Ratio of Late packets for IP CS CN (%)],

  -- 12. Media Stream Channel Utilization Rate (%) [Daily metric joined on NE_NAME & DATE_ID]
  CAST(ROUND(
    CASE 
      WHEN COALESCE(MSC.maxNrOfLicMediaStreamChannels, 0) = 0 THEN 0.0
      WHEN ((MSC.pmNrOfMediaStreamChannelsBusy * 100.0) / MSC.maxNrOfLicMediaStreamChannels) > 100.0 THEN 100.0
      ELSE (MSC.pmNrOfMediaStreamChannelsBusy * 100.0) / MSC.maxNrOfLicMediaStreamChannels
    END, 1) AS NUMERIC(10,1)) AS [Media Stream Channel Utilization Rate (%)]

FROM (
  -- Master Spine of Keys (Hourly Granularity)
  SELECT 
    NE_NAME, DATE_ID, HOUR_ID, MAX(DC_RELEASE) AS NE_VERSION
  FROM (
    SELECT DISTINCT MGW.NE_NAME, DT.DATE_ID, TM.HOUR_ID, T.DC_RELEASE
    FROM DC.DIM_E_CN_MGW MGW
    JOIN DC.DC_E_MGW_JITTERHANDLINGSRV_COUNT T ON T.MGW = MGW.NE_ID AND T.OSS_ID = MGW.OSS_ID
    JOIN DC.DIM_DATE DT ON T.DATE_ID = DT.DATE_ID
    JOIN DC.DIM_TIME TM ON T.HOUR_ID = TM.HOUR_ID AND T.MIN_ID = TM.MIN_ID
    JOIN DC.DIM_ROWSTATUS RS ON T.ROWSTATUS = RS.ROWSTATUS
    WHERE RS.ROWSTATUS NOT IN ('DUPLICATE', 'SUSPECTED')
      AND MGW.NE_NAME IN ('ARMGW01','ARMGW02','ARMGW03','ARMGW04','ARMGW05','ARMGW06','BBMGW01','BBMGW02','BSMGW1','DUMGW01','DUMGW02','KIMGW01','SUMGW01','SUMGW02','TBGMGW1','TEMGW1','TRMGW01','TRMGW02')
      AND DT.DATE_ID BETWEEN DATEADD(DAY, -30, NOW()) AND NOW()
    UNION
    SELECT DISTINCT MGW.NE_NAME, DT.DATE_ID, TM.HOUR_ID, T.DC_RELEASE
    FROM DC.DIM_E_CN_MGW MGW
    JOIN DC.DC_E_MGW_IPACCESSHOSTGPB_COUNT T ON T.MGW = MGW.NE_ID AND T.OSS_ID = MGW.OSS_ID
    JOIN DC.DIM_DATE DT ON T.DATE_ID = DT.DATE_ID
    JOIN DC.DIM_TIME TM ON T.HOUR_ID = TM.HOUR_ID AND T.MIN_ID = TM.MIN_ID
    JOIN DC.DIM_ROWSTATUS RS ON T.ROWSTATUS = RS.ROWSTATUS
    WHERE RS.ROWSTATUS NOT IN ('DUPLICATE', 'SUSPECTED')
      AND MGW.NE_NAME IN ('ARMGW01','ARMGW02','ARMGW03','ARMGW04','ARMGW05','ARMGW06','BBMGW01','BBMGW02','BSMGW1','DUMGW01','DUMGW02','KIMGW01','SUMGW01','SUMGW02','TBGMGW1','TEMGW1','TRMGW01','TRMGW02')
      AND DT.DATE_ID BETWEEN DATEADD(DAY, -30, NOW()) AND NOW()
    UNION
    SELECT DISTINCT MGW.NE_NAME, DT.DATE_ID, TM.HOUR_ID, T.DC_RELEASE
    FROM DC.DIM_E_CN_MGW MGW
    JOIN DC.DC_E_MGW_MGWAPPLICATION_COUNT T ON T.MGW = MGW.NE_ID AND T.OSS_ID = MGW.OSS_ID
    JOIN DC.DIM_DATE DT ON T.DATE_ID = DT.DATE_ID
    JOIN DC.DIM_TIME TM ON T.HOUR_ID = TM.HOUR_ID AND T.MIN_ID = TM.MIN_ID
    JOIN DC.DIM_ROWSTATUS RS ON T.ROWSTATUS = RS.ROWSTATUS
    WHERE RS.ROWSTATUS NOT IN ('DUPLICATE', 'SUSPECTED')
      AND MGW.NE_NAME IN ('ARMGW01','ARMGW02','ARMGW03','ARMGW04','ARMGW05','ARMGW06','BBMGW01','BBMGW02','BSMGW1','DUMGW01','DUMGW02','KIMGW01','SUMGW01','SUMGW02','TBGMGW1','TEMGW1','TRMGW01','TRMGW02')
      AND DT.DATE_ID BETWEEN DATEADD(DAY, -30, NOW()) AND NOW()
    UNION
    SELECT DISTINCT MGW.NE_NAME, DT.DATE_ID, TM.HOUR_ID, NULL AS DC_RELEASE
    FROM DC.DIM_E_CN_MGW MGW
    JOIN DC.DC_E_MGW_REMOTESITE_COUNT T ON T.MGW = MGW.NE_ID AND T.OSS_ID = MGW.OSS_ID
    JOIN DC.DIM_DATE DT ON T.DATE_ID = DT.DATE_ID
    JOIN DC.DIM_TIME TM ON T.HOUR_ID = TM.HOUR_ID AND T.MIN_ID = TM.MIN_ID
    JOIN DC.DIM_ROWSTATUS RS ON T.ROWSTATUS = RS.ROWSTATUS
    WHERE RS.ROWSTATUS NOT IN ('DUPLICATE', 'SUSPECTED')
      AND MGW.NE_NAME IN ('ARMGW01','ARMGW02','ARMGW03','ARMGW04','ARMGW05','ARMGW06','BBMGW01','BBMGW02','BSMGW1','DUMGW01','DUMGW02','KIMGW01','SUMGW01','SUMGW02','TBGMGW1','TEMGW1','TRMGW01','TRMGW02')
      AND DT.DATE_ID BETWEEN DATEADD(DAY, -30, NOW()) AND NOW()
    UNION
    SELECT DISTINCT MGW.NE_NAME, DT.DATE_ID, TM.HOUR_ID, NULL AS DC_RELEASE
    FROM DC.DIM_E_CN_MGW MGW
    JOIN DC.DC_E_MGW_UNKNOWNREMOTESITE_COUNT T ON T.MGW = MGW.NE_ID AND T.OSS_ID = MGW.OSS_ID
    JOIN DC.DIM_DATE DT ON T.DATE_ID = DT.DATE_ID
    JOIN DC.DIM_TIME TM ON T.HOUR_ID = TM.HOUR_ID AND T.MIN_ID = TM.MIN_ID
    JOIN DC.DIM_ROWSTATUS RS ON T.ROWSTATUS = RS.ROWSTATUS
    WHERE RS.ROWSTATUS NOT IN ('DUPLICATE', 'SUSPECTED')
      AND MGW.NE_NAME IN ('ARMGW01','ARMGW02','ARMGW03','ARMGW04','ARMGW05','ARMGW06','BBMGW01','BBMGW02','BSMGW1','DUMGW01','DUMGW02','KIMGW01','SUMGW01','SUMGW02','TBGMGW1','TEMGW1','TRMGW01','TRMGW02')
      AND DT.DATE_ID BETWEEN DATEADD(DAY, -30, NOW()) AND NOW()
  ) RAW_KEYS
  GROUP BY NE_NAME, DATE_ID, HOUR_ID
) S

-- 1. Jitter Handling Service (Hourly Aggregation)
LEFT JOIN (
  SELECT
    MGW.NE_NAME, DT.DATE_ID, TM.HOUR_ID,
    SUM(CASE WHEN T.pmIpCnConnMeasuredJitter0 < 0 THEN 0 ELSE T.pmIpCnConnMeasuredJitter0 END) AS pmIpCnConnMeasuredJitter0,
    SUM(CASE WHEN T.pmIpCnConnMeasuredJitter1 < 0 THEN 0 ELSE T.pmIpCnConnMeasuredJitter1 END) AS pmIpCnConnMeasuredJitter1,
    SUM(CASE WHEN T.pmIpCnConnMeasuredJitter2 < 0 THEN 0 ELSE T.pmIpCnConnMeasuredJitter2 END) AS pmIpCnConnMeasuredJitter2,
    SUM(CASE WHEN T.pmIpCnConnMeasuredJitter3 < 0 THEN 0 ELSE T.pmIpCnConnMeasuredJitter3 END) AS pmIpCnConnMeasuredJitter3,
    SUM(CASE WHEN T.pmIpCnConnMeasuredJitter4 < 0 THEN 0 ELSE T.pmIpCnConnMeasuredJitter4 END) AS pmIpCnConnMeasuredJitter4,
    SUM(CASE WHEN T.pmIpCnConnMeasuredJitter5 < 0 THEN 0 ELSE T.pmIpCnConnMeasuredJitter5 END) AS pmIpCnConnMeasuredJitter5,
    SUM(CASE WHEN T.pmVoIpConnMeasuredJitter0 < 0 THEN 0 ELSE T.pmVoIpConnMeasuredJitter0 END) AS pmVoIpConnMeasuredJitter0,
    SUM(CASE WHEN T.pmVoIpConnMeasuredJitter1 < 0 THEN 0 ELSE T.pmVoIpConnMeasuredJitter1 END) AS pmVoIpConnMeasuredJitter1,
    SUM(CASE WHEN T.pmVoIpConnMeasuredJitter2 < 0 THEN 0 ELSE T.pmVoIpConnMeasuredJitter2 END) AS pmVoIpConnMeasuredJitter2,
    SUM(CASE WHEN T.pmVoIpConnMeasuredJitter3 < 0 THEN 0 ELSE T.pmVoIpConnMeasuredJitter3 END) AS pmVoIpConnMeasuredJitter3,
    SUM(CASE WHEN T.pmVoIpConnMeasuredJitter4 < 0 THEN 0 ELSE T.pmVoIpConnMeasuredJitter4 END) AS pmVoIpConnMeasuredJitter4,
    SUM(CASE WHEN T.pmVoIpConnMeasuredJitter5 < 0 THEN 0 ELSE T.pmVoIpConnMeasuredJitter5 END) AS pmVoIpConnMeasuredJitter5,
    SUM(CASE WHEN T.pmVoIpConnMeasuredJitter6 < 0 THEN 0 ELSE T.pmVoIpConnMeasuredJitter6 END) AS pmVoIpConnMeasuredJitter6,
    SUM(CASE WHEN T.pmVoIpConnMeasuredJitter7 < 0 THEN 0 ELSE T.pmVoIpConnMeasuredJitter7 END) AS pmVoIpConnMeasuredJitter7,
    SUM(CASE WHEN T.pmVoIpConnMeasuredJitter8 < 0 THEN 0 ELSE T.pmVoIpConnMeasuredJitter8 END) AS pmVoIpConnMeasuredJitter8,
    SUM(CASE WHEN T.pmIpRanConnMeasuredJitter0 < 0 THEN 0 ELSE T.pmIpRanConnMeasuredJitter0 END) AS pmIpRanConnMeasuredJitter0,
    SUM(CASE WHEN T.pmIpRanConnMeasuredJitter1 < 0 THEN 0 ELSE T.pmIpRanConnMeasuredJitter1 END) AS pmIpRanConnMeasuredJitter1,
    SUM(CASE WHEN T.pmIpRanConnMeasuredJitter2 < 0 THEN 0 ELSE T.pmIpRanConnMeasuredJitter2 END) AS pmIpRanConnMeasuredJitter2,
    SUM(CASE WHEN T.pmIpRanConnMeasuredJitter3 < 0 THEN 0 ELSE T.pmIpRanConnMeasuredJitter3 END) AS pmIpRanConnMeasuredJitter3,
    SUM(CASE WHEN T.pmIpRanConnMeasuredJitter4 < 0 THEN 0 ELSE T.pmIpRanConnMeasuredJitter4 END) AS pmIpRanConnMeasuredJitter4,
    SUM(CASE WHEN T.pmIpRanConnMeasuredJitter5 < 0 THEN 0 ELSE T.pmIpRanConnMeasuredJitter5 END) AS pmIpRanConnMeasuredJitter5,
    SUM(CASE WHEN T.pmLatePktsIpCn < 0 THEN 0 ELSE T.pmLatePktsIpCn END) AS pmLatePktsIpCn,
    SUM(CASE WHEN T.pmSuccTransmittedPktsIpCn < 0 THEN 0 ELSE T.pmSuccTransmittedPktsIpCn END) AS pmSuccTransmittedPktsIpCn,
    SUM(CASE WHEN T.pmLatePktsExt < 0 THEN 0 ELSE T.pmLatePktsExt END) AS pmLatePktsExt,
    SUM(CASE WHEN T.pmSuccTransmittedPktsExtLo < 0 THEN 0 ELSE T.pmSuccTransmittedPktsExtLo END) AS pmSuccTransmittedPktsExtLo,
    SUM(CASE WHEN T.pmSuccTransmittedPktsExtHi < 0 THEN 0 ELSE T.pmSuccTransmittedPktsExtHi END) AS pmSuccTransmittedPktsExtHi,
    SUM(CASE WHEN T.pmLatePktsIpRan < 0 THEN 0 ELSE T.pmLatePktsIpRan END) AS pmLatePktsIpRan,
    SUM(CASE WHEN T.pmSuccTransmittedPktsIpRan < 0 THEN 0 ELSE T.pmSuccTransmittedPktsIpRan END) AS pmSuccTransmittedPktsIpRan
  FROM DC.DIM_E_CN_MGW MGW
  JOIN DC.DC_E_MGW_JITTERHANDLINGSRV_COUNT T ON T.MGW = MGW.NE_ID AND T.OSS_ID = MGW.OSS_ID
  JOIN DC.DIM_DATE DT ON T.DATE_ID = DT.DATE_ID
  JOIN DC.DIM_TIME TM ON T.HOUR_ID = TM.HOUR_ID AND T.MIN_ID = TM.MIN_ID
  JOIN DC.DIM_ROWSTATUS RS ON T.ROWSTATUS = RS.ROWSTATUS
  WHERE RS.ROWSTATUS NOT IN ('DUPLICATE', 'SUSPECTED')
    AND MGW.NE_NAME IN ('ARMGW01','ARMGW02','ARMGW03','ARMGW04','ARMGW05','ARMGW06','BBMGW01','BBMGW02','BSMGW1','DUMGW01','DUMGW02','KIMGW01','SUMGW01','SUMGW02','TBGMGW1','TEMGW1','TRMGW01','TRMGW02')
    AND DT.DATE_ID BETWEEN DATEADD(DAY, -30, NOW()) AND NOW()
  GROUP BY MGW.NE_NAME, DT.DATE_ID, TM.HOUR_ID
) J ON S.NE_NAME = J.NE_NAME AND S.DATE_ID = J.DATE_ID AND S.HOUR_ID = J.HOUR_ID

-- 2. Data Coverage / Availability
LEFT JOIN (
  SELECT
    MGW.NE_NAME, DT.DATE_ID,
    AVG(T.DATACOVERAGE) AS DATACOVERAGE,
    AVG(T.PERIOD_DURATION) AS PERIOD_DURATION
  FROM DC.DIM_E_CN_MGW MGW
  JOIN DC.DC_E_MGW_JITTERHANDLINGSRV_DAY T ON T.MGW = MGW.NE_ID AND T.OSS_ID = MGW.OSS_ID
  JOIN DC.DIM_DATE DT ON T.DATE_ID = DT.DATE_ID
  JOIN DC.DIM_ROWSTATUS RS ON T.ROWSTATUS = RS.ROWSTATUS
  WHERE RS.ROWSTATUS NOT IN ('DUPLICATE', 'SUSPECTED')
    AND MGW.NE_NAME IN ('ARMGW01','ARMGW02','ARMGW03','ARMGW04','ARMGW05','ARMGW06','BBMGW01','BBMGW02','BSMGW1','DUMGW01','DUMGW02','KIMGW01','SUMGW01','SUMGW02','TBGMGW1','TEMGW1','TRMGW01','TRMGW02')
    AND DT.DATE_ID BETWEEN DATEADD(DAY, -30, NOW()) AND NOW()
  GROUP BY MGW.NE_NAME, DT.DATE_ID
) DA ON S.NE_NAME = DA.NE_NAME AND S.DATE_ID = DA.DATE_ID

-- 3. Remote Site R4 (Hourly Aggregation)
LEFT JOIN (
  SELECT
    MGW.NE_NAME, DT.DATE_ID, TM.HOUR_ID,
    SUM(CASE WHEN T.pmRtpDiscardedPkts < 0 THEN 0 ELSE T.pmRtpDiscardedPkts END) AS pmRtpDiscardedPkts,
    SUM(CASE WHEN T.pmRtpLostPkts < 0 THEN 0 ELSE T.pmRtpLostPkts END) AS pmRtpLostPkts,
    SUM(CASE WHEN T.pmRtpReceivedPkts < 0 THEN 0 ELSE T.pmRtpReceivedPkts END) AS pmRtpReceivedPkts
  FROM DC.DIM_E_CN_MGW MGW
  JOIN DC.DC_E_MGW_REMOTESITE_COUNT T ON T.MGW = MGW.NE_ID AND T.OSS_ID = MGW.OSS_ID
  JOIN DC.DIM_DATE DT ON T.DATE_ID = DT.DATE_ID
  JOIN DC.DIM_TIME TM ON T.HOUR_ID = TM.HOUR_ID AND T.MIN_ID = TM.MIN_ID
  JOIN DC.DIM_ROWSTATUS RS ON T.ROWSTATUS = RS.ROWSTATUS
  WHERE T.DC_RELEASE LIKE 'R4.%'
    AND RS.ROWSTATUS NOT IN ('DUPLICATE', 'SUSPECTED')
