-- Unified MSC KPI Report - Complete Script
-- Fixed bugs: Date comparisons, division by zero, joins, column names, duplicates, and CSSR calculation
-- Compatible with Sybase IQ
-- Aggregated to DAY level (HOUR_ID removed)

WITH 
MSC_AGG AS (
    SELECT ELEM, DATE_ID,
        SUM(SECHAND_NAUTREQSUCC) AS NAUTREQSUCC, SUM(SECHAND_NAUTREQTOT) AS NAUTREQTOT,
        SUM(CHASSIGNT_NCHAFRMSUCC) AS NCHAFRMSUCC, SUM(CHASSIGNT_NCHAFRMTOT) AS NCHAFRMTOT,
        SUM(CHASSIGNT_NCHATOMSUCC) AS NCHATOMSUCC, SUM(CHASSIGNT_NCHATOMTOT) AS NCHATOMTOT,
        SUM(SECHAND_NCIPSETSUCC) AS NCIPSETSUCC, SUM(SECHAND_NCIPATTTOT) AS NCIPATTTOT,
        SUM(UMTSSEC_NSECFRRNSCC) AS NSECFRRNSCC, SUM(UMTSSEC_NSECTORNTOT) AS NSECTORNTOT,
        SUM(EQIDCON_NEQIDMSSUCC) AS NEQIDMSSUCC, SUM(EQIDCON_NEQIDTOT) AS NEQIDTOT,
        SUM(UPDLOCAT_NLOCNRGSUCC) AS NLOCNRGSUCC, SUM(UPDLOCAT_NLOCNRGTOT) AS NLOCNRGTOT,
        SUM(UPDLOCAT_NLOCOLDSUCC) AS NLOCOLDSUCC, SUM(UPDLOCAT_NLOCOLDTOT) AS NLOCOLDTOT,
        SUM(UPDLOCAT2_NLLOCOLDTOT) AS NLLOCOLDTOT, SUM(UPDALOCAT_NLOCAROLDSUCC) AS NLOCAROLDSUCC,
        SUM(UPDALOCAT_NLOCAOLDTOT) AS NLOCAOLDTOT, SUM(UPDALOCAT_NLALOCOLDTOT) AS NLALOCOLDTOT,
        SUM(UPDALOCAT_NLNAPERSUCC) AS NLNAPERSUCC, SUM(UPDLOCAT_NLOCPERTOT) AS NLOCPERTOT,
        SUM(UPDLOCAT2_NLLOCPERTOT) AS NLLOCPERTOT, SUM(UPDLOCAT_NLOCNRG2SUCC) AS NLOCNRG2SUCC,
        SUM(UPDLOCAT_NLOCOLD2SUCC) AS NLOCOLD2SUCC, SUM(UPDLOCAT_NLOCNRG2TOT) AS NLOCNRG2TOT,
        SUM(UPDLOCAT_NLOCOLD2TOT) AS NLOCOLD2TOT,
        SUM(PAGING_NPAG1RESUCC) AS NPAG1RESUCC, SUM(PAGING_NPAG2RESUCC) AS NPAG2RESUCC,
        SUM(PAGING_NPAG1REUSUCC) AS NPAG1REUSUCC, SUM(PAGING_NPAG2REUSUCC) AS NPAG2REUSUCC,
        SUM(PAGING_NPAG1LOTOT) AS NPAG1LOTOT, SUM(PAGING_NPAG1LOUTOT) AS NPAG1LOUTOT,
        SUM(PAGING_NPAG2LOTOT) AS NPAG2LOTOT, SUM(PAGING_NPAG2LOUTOT) AS NPAG2LOUTOT,
        SUM(PAGING_NPAG1GLTOT) AS NPAG1GLTOT, SUM(PAGING_NPAG2GLTOT) AS NPAG2GLTOT,
        SUM(PAGING_NPAG1GLUTOT) AS NPAG1GLUTOT, SUM(PAGING_NPAG2GLUTOT) AS NPAG2GLUTOT,
        SUM(PAGING_NPAGIUREA) AS NPAGIUREA, SUM(PAGING_NPAGAREIU) AS NPAGAREIU,
        SUM(CHASSIGNT_NMSFRMSCCI) AS NMSFRMSCCI, SUM(CHASSIGNT_NMSFRMTOTI) AS NMSFRMTOTI,
        SUM(CHASSIGNT_NMSTOMSCCO) AS NMSTOMSCCO, SUM(CHASSIGNT_NMSTOMTOTO) AS NMSTOMTOTO,
        SUM(SHMSGSERV_NSMSCAOSUCC) AS NSMSCAOSUCC, SUM(SHMSGSERV_NSMSCDOTOT) AS NSMSCDOTOT,
        SUM(SHMSGSERV_NSMSSRSUCC) AS NSMSSRSUCC, SUM(SHMSGSERV_NSMSSMRLTOT) AS NSMSSMRLTOT,
        SUM(SHMSGSERV_NSMSCMRSUCC) AS NSMSCMRSUCC, SUM(SHMSGSERV_NSMSCMTOT) AS NSMSCMTOT,
        SUM(SHMSGSERV_NSMSRAOSUCC) AS NSMSRAOSUCC, SUM(SHMSGSERV_NSMSRDOTOT) AS NSMSRDOTOT,
        SUM(PRBTCALL_PRBTSUCC) AS PRBTSUCC, SUM(PRBTCALL_PRBTTOT) AS PRBTTOT,
        SUM(GUHNDOVER_NGUHBSCRNCSUCC) AS IntraMSC_G2U_SUCC, SUM(GUHNDOVER_NGUHBSCRNCTOT) AS IntraMSC_G2U_TOT,
        SUM(UGHNDOVER_NUGHRNCBSCSUCC) AS IntraMSC_U2G_SUCC, SUM(UGHNDOVER_NUGHRNCBSCTOT) AS IntraMSC_U2G_TOT,
        SUM(L3CCMSG_NL3CPROC) AS NL3CPROC
    FROM DC.DC_E_CNAXE_MSC_DAY
    WHERE CAST(DATE_ID AS DATE) >= CAST(GETDATE() - 180 AS DATE)
      AND ELEM IN ('ARBC01','ARBC02','ARMSS5','BGBC01','BGBC02','BSMSS2')
    GROUP BY ELEM, DATE_ID
),
MTRAFTYPE_AGG AS (
    SELECT ELEM, DATE_ID,
        SUM(NTRALACC) AS ERLANG, SUM(NUNSUCC) AS NUNSUCC,
        SUM(NICONG) AS NICONG, SUM(NECONG) AS NECONG, SUM(NCALLS) AS NCALLS
    FROM DC.DC_E_CNAXE_MTRAFTYPE_DAY
    WHERE CAST(DATE_ID AS DATE) >= CAST(GETDATE() - 180 AS DATE)
      AND ELEM IN ('ARBC01','ARBC02','ARMSS5','BGBC01','BGBC02','BSMSS2')
    GROUP BY ELEM, DATE_ID
),
TRUNKROUTE_AGG AS (
    SELECT ELEM, DATE_ID, SUM(NCALLSI) AS NCALLSI
    FROM DC.DC_E_CNAXE_TRUNKROUTE_DAY
    WHERE CAST(DATE_ID AS DATE) >= CAST(GETDATE() - 180 AS DATE)
      AND ELEM IN ('ARBC01','ARBC02','ARMSS5','BGBC01','BGBC02','BSMSS2')
    GROUP BY ELEM, DATE_ID
),
STS_TOT_AGG AS (
    SELECT ELEM, DATE_ID,
        SUM(LOAS_NFTDORG) AS LOAS_NFTDORG, SUM(LOAS_NFTDIEX) AS LOAS_NFTDIEX,
        SUM(LOAS_NOFFORG) AS LOAS_NOFFORG, SUM(LOAS_NOFFIEX) AS LOAS_NOFFIEX
    FROM DC.DC_E_CMN_STS_TOT_DAY
    WHERE CAST(DATE_ID AS DATE) >= CAST(GETDATE() - 180 AS DATE)
      AND ELEM IN ('ARBC01','ARBC02','ARMSS5','BGBC01','BGBC02','BSMSS2')
    GROUP BY ELEM, DATE_ID
),
SPXLC_AGG AS (
    SELECT ELEM, DATE_ID,
        SUM(NFTORG) AS NFTORG, SUM(NFTIEX) AS NFTIEX, SUM(SOROF) AS SOROF,
        SUM(SINOF) AS SINOF, SUM(SORRE) AS SORRE, SUM(SINRE) AS SINRE
    FROM DC.DC_E_CNAXE_SPXLC_DAY
    WHERE CAST(DATE_ID AS DATE) >= CAST(GETDATE() - 180 AS DATE)
      AND ELEM IN ('ARBC01','ARBC02','ARMSS5','BGBC01','BGBC02','BSMSS2')
    GROUP BY ELEM, DATE_ID
),
SGS_AGG AS (
    SELECT ELEM, DATE_ID,
        SUM(SGS_NSPAG1SGS) AS NSPAG1SGS, SUM(SGS_NSPAG2SGS) AS NSPAG2SGS,
        SUM(SGS_NTPAG1SGS) AS NTPAG1SGS, SUM(SGS_NTPAG2SGS) AS NTPAG2SGS,
        SUM(CSFB_NSUCCCSFB) AS NSUCCCSFB, SUM(CSFB_NUNSUCCCSFB) AS NUNSUCCCSFB,
        SUM(CSFB_NUSREJCSFB) AS NUSREJCSFB, SUM(CSFB_NSPAG1CSFB) AS NSPAG1CSFB,
        SUM(CSFB_NSPAG2CSFB) AS NSPAG2CSFB, SUM(CSFB_NTPAG1CSFB) AS NTPAG1CSFB,
        SUM(CSFB_NTPAG2CSFB) AS NTPAG2CSFB, SUM(SGS_NSLOCREGSGS) AS NSLOCREGSGS,
        SUM(SGS_NSLOCNREGSGS) AS NSLOCNREGSGS, SUM(SGS_NTLOCREGSGS) AS NTLOCREGSGS,
        SUM(SGS_NTLOCNREGSGS) AS NTLOCNREGSGS
    FROM DC.DC_E_CNAXE_SGCSFB_DAY
    WHERE CAST(DATE_ID AS DATE) >= CAST(GETDATE() - 180 AS DATE)
      AND ELEM IN ('ARBC01','ARBC02','ARMSS5','BGBC01','BGBC02','BSMSS2')
    GROUP BY ELEM, DATE_ID
),
MSCOBJ_AGG AS (
    SELECT ELEM, DATE_ID,
        SUM(NBRMSCLST_NNBRHBANSUCC + NBRMSCLST_NNBRHSANSUCC + NBRMSCLST_NNBRHINASUCC +
            NBRMSCLST_NNBRHSNASUCC + NBRMSCLST_NNBRHTHISUCC) AS InterMSC_GSM_SUCC,
        SUM(NBRMSCLST_NNBRHBANTOT + NBRMSCLST_NNBRHSANTOT + NBRMSCLST_NNBRHINATOT +
            NBRMSCLST_NNBRHSNATOT + NBRMSCLST_NNBRHTHITOT) AS InterMSC_GSM_TOT,
        SUM(NBRMSCUGHO_NNBRBUGASUCC + NBRMSCUGHO_NNBRSUGSUCC + NBRMSCUGHO_NNBRSIHOATCHSUCC +
            NBRMSCUGHO_NNBRITCHHOASUCC) AS InterMSC_U2G_SUCC,
        SUM(NBRMSCUGHO_NNBRBUGATOT + NBRMSCUGHO_NNBRSUGTOT + NBRMSCUGHO_NNBRIHOATCHTOT +
            NBRMSCUGHO_NNBRISDHHOATOT) AS InterMSC_U2G_TOT
    FROM DC.DC_E_CNAXE_MSCOBJ_DAY
    WHERE CAST(DATE_ID AS DATE) >= CAST(GETDATE() - 180 AS DATE)
      AND ELEM IN ('ARBC01','ARBC02','ARMSS5','BGBC01','BGBC02','BSMSS2')
    GROUP BY ELEM, DATE_ID
),
MAP_AGG AS (
    SELECT RAW.ELEM, RAW.DATE_ID,
        SUM(COALESCE(RAW.NMAP1SUCC,0)) AS NMAP1SUCC, SUM(COALESCE(RAW.NMAP2SUCC,0)) AS NMAP2SUCC,
        SUM(COALESCE(RAW.NMAP3SUCC,0)) AS NMAP3SUCC, SUM(COALESCE(RAW.NMAP1TOT,0)) AS NMAP1TOT,
        SUM(COALESCE(RAW.NMAP2TOT,0)) AS NMAP2TOT, SUM(COALESCE(RAW.NMAP3TOT,0)) AS NMAP3TOT,
        SUM(COALESCE(RAW.NMAP1TCT,0)) AS NMAP1TCT
    FROM DC.DC_E_CNAXE_MAPSIGIWRK_DAY RAW
    JOIN DC.DIM_ROWSTATUS RS ON RAW.ROWSTATUS = RS.ROWSTATUS
    WHERE CAST(RAW.DATE_ID AS DATE) >= CAST(GETDATE() - 180 AS DATE)
      AND RS.ROWSTATUS NOT IN ('DUPLICATE','SUSPECTED')
      AND RAW.ELEM IN ('ARBC01','ARBC02','ARMSS5','BGBC01','BGBC02','BSMSS2')
    GROUP BY RAW.ELEM, RAW.DATE_ID
),
RNC_AGG AS (
    SELECT ELEM, DATE_ID,
        SUM(NRNFRMSCCI) AS NRNFRMSCCI, SUM(NRNTOMSCCO) AS NRNTOMSCCO,
        SUM(NRNFRMTOTI) AS NRNFRMTOTI, SUM(NRNTOMTOTO) AS NRNTOMTOTO
    FROM DC.DC_E_CNAXE_RNCSTAT_DAY
    WHERE CAST(DATE_ID AS DATE) >= CAST(GETDATE() - 180 AS DATE)
      AND ELEM IN ('ARBC01','ARBC02','ARMSS5','BGBC01','BGBC02','BSMSS2')
    GROUP BY ELEM, DATE_ID
),
VLR_AGG AS (
    SELECT R.ELEM, R.DATE_ID,
        SUM(R.NSUB) AS VLR_Subs_Registered, SUM(R.NSUBA) AS VLR_Subs_Active
    FROM DC.DC_E_CNAXE_VLRSUBSCR_DAY R
    JOIN DC.DIM_ROWSTATUS S ON R.ROWSTATUS = S.ROWSTATUS
    WHERE CAST(R.DATE_ID AS DATE) >= CAST(GETDATE() - 180 AS DATE)
      AND S.ROWSTATUS NOT IN ('DUPLICATE','SUSPECTED')
      AND R.ELEM IN ('ARBC01','ARBC02','ARMSS5','BGBC01','BGBC02','BSMSS2')
    GROUP BY R.ELEM, R.DATE_ID
),
NE_INFO AS (
    SELECT NE_NAME, NE_ID, SOURCE_TYPE, NE_VERSION
    FROM (
        SELECT 
            NE_NAME, NE_ID, SOURCE_TYPE, NE_VERSION,
            ROW_NUMBER() OVER(PARTITION BY NE_NAME ORDER BY NE_VERSION DESC) as rn
        FROM DC.DIM_E_CN_AXE
        WHERE NE_NAME IN ('ARBC01','ARBC02','ARMSS5','BGBC01','BGBC02','BSMSS2')
          AND (BLADE_NAME IS NULL OR BLADE_NAME = '')
    ) ranked
    WHERE rn = 1
)

SELECT
    M.ELEM AS node,
    CAST(M.DATE_ID AS DATE) AS day,
    
    -- Authentication
    100.0 * M.NAUTREQSUCC / NULLIF(M.NAUTREQTOT, 0) AS authentication_sr,
    
    -- Channel Assignment
    100.0 * M.NMSFRMSCCI / NULLIF(M.NMSFRMTOTI, 0) AS mo_cassr,
    100.0 * M.NMSTOMSCCO / NULLIF(M.NMSTOMTOTO, 0) AS mt_cassr,
    100.0 * (M.NCHAFRMSUCC + M.NCHATOMSUCC) / NULLIF(M.NCHAFRMTOT + M.NCHATOMTOT, 0) AS cassr_gsm,
    
    -- Ciphering
    100.0 * M.NCIPSETSUCC / NULLIF(M.NCIPATTTOT, 0) AS ciphering_sr_gsm,
    100.0 * M.NSECFRRNSCC / NULLIF(M.NSECTORNTOT, 0) AS ciphering_sr_wcdma,
    
    -- IMEI
    100.0 * M.NEQIDMSSUCC / NULLIF(M.NEQIDTOT, 0) AS imei_fetch_ms_sr,
    
    -- Location Update
    100.0 * M.NLOCNRGSUCC / NULLIF(M.NLOCNRGTOT, 0) AS non_registered_subscribers_lusr,
    100.0 * M.NLOCOLDSUCC / NULLIF(M.NLOCOLDTOT - M.NLLOCOLDTOT, 0) AS registered_subscribers_lusr,
    100.0 * M.NLOCAROLDSUCC / NULLIF(M.NLOCAOLDTOT - M.NLALOCOLDTOT, 0) AS registered_gsm_subscribers_lusr,
    100.0 * M.NLNAPERSUCC / NULLIF(M.NLOCPERTOT - M.NLLOCPERTOT, 0) AS periodic_update_lusr,
    100.0 * SGS.NSLOCREGSGS / NULLIF(SGS.NTLOCREGSGS, 0) AS registered_sgs_subscriber_lusr,
    100.0 * SGS.NSLOCNREGSGS / NULLIF(SGS.NTLOCNREGSGS, 0) AS non_registered_sgs_subscriber_lusr,
    
    -- Overall LUSR (version dependent)
    CASE 
        WHEN NE.NE_VERSION LIKE 'R1%' AND NE.NE_VERSION < 'R14.1' THEN
            100.0 * (M.NLOCNRGSUCC + M.NLOCOLDSUCC + M.NLOCNRG2SUCC + M.NLOCOLD2SUCC) /
                    NULLIF(M.NLOCNRGTOT + M.NLOCOLDTOT + M.NLOCNRG2TOT + M.NLOCOLD2TOT, 0)
        ELSE
            100.0 * (M.NLOCNRGSUCC + M.NLOCOLDSUCC + M.NLOCNRG2SUCC + M.NLOCOLD2SUCC +
                     SGS.NSLOCREGSGS + SGS.NSLOCNREGSGS) /
                    NULLIF(M.NLOCNRGTOT + M.NLOCOLDTOT + M.NLOCNRG2TOT + M.NLOCOLD2TOT -
                           M.NLLOCOLDTOT + SGS.NTLOCREGSGS + SGS.NTLOCNREGSGS, 0)
    END AS overall_lusr,
    
    -- Paging
    100.0 * (M.NPAG1RESUCC + M.NPAG2RESUCC + M.NPAG1REUSUCC + M.NPAG2REUSUCC) /
        NULLIF(M.NPAG1LOTOT + M.NPAG1LOUTOT + M.NPAG2LOTOT + M.NPAG2LOUTOT + M.NPAG1GLTOT + M.NPAG2GLTOT, 0) 
        AS system_perceived_paging_success,
    100.0 * (M.NPAG1RESUCC + M.NPAG2RESUCC) / NULLIF(M.NPAG1GLTOT + M.NPAG1LOTOT, 0) AS paging_success_sr,
    100.0 * (M.NPAG1RESUCC + M.NPAG2RESUCC + M.NPAGIUREA) / NULLIF(M.NPAG1LOTOT + M.NPAG1GLTOT, 0) AS gsm_paging_end_user_sr,
    100.0 * (M.NPAG1REUSUCC + M.NPAG2REUSUCC + M.NPAGAREIU) / NULLIF(M.NPAG1LOUTOT + M.NPAG1GLUTOT, 0) AS wcdma_paging_end_user_sr,
    
    -- RAB Assignment
    100.0 * (RNC.NRNFRMSCCI + RNC.NRNTOMSCCO) / NULLIF(RNC.NRNFRMTOTI + RNC.NRNTOMTOTO, 0) AS rab_assignment_sr,
    
    -- SMS
    100.0 * M.NSMSCAOSUCC / NULLIF(M.NSMSCDOTOT, 0) AS msc_originating_sms_sr,
    100.0 * M.NSMSRAOSUCC / NULLIF(M.NSMSRDOTOT, 0) AS originating_sms_sr,
    100.0 * M.NSMSSRSUCC / NULLIF(M.NSMSSMRLTOT, 0) AS terminating_sms_sr,
    100.0 * M.NSMSCMRSUCC / NULLIF(M.NSMSCMTOT, 0) AS msc_terminating_sms_sr,  
     
    -- SGs/CSFB
    100.0 * (SGS.NSPAG1SGS + SGS.NSPAG2SGS) / NULLIF(SGS.NTPAG1SGS, 0) AS sms_sgs_end_user_paging_success,
    100.0 * (SGS.NSPAG1SGS + SGS.NSPAG2SGS) / NULLIF(SGS.NTPAG1SGS + SGS.NTPAG2SGS, 0) AS sms_sgs_system_paging_success,
    100.0 * SGS.NSUCCCSFB / NULLIF(SGS.NSUCCCSFB + SGS.NUNSUCCCSFB + SGS.NUSREJCSFB, 0) AS csfb_sgs_success,
    100.0 * SGS.NSUCCCSFB / NULLIF(SGS.NSUCCCSFB + SGS.NUNSUCCCSFB, 0) AS csfb_sgs_success_network_performance,
    
    -- Handover
    100.0 * MSCOBJ.InterMSC_GSM_SUCC / NULLIF(MSCOBJ.InterMSC_GSM_TOT, 0) AS inter_msc_handover_success_gsm,
    100.0 * MSCOBJ.InterMSC_U2G_SUCC / NULLIF(MSCOBJ.InterMSC_U2G_TOT, 0) AS inter_msc_handover_success_u2g,
    100.0 * M.IntraMSC_G2U_SUCC / NULLIF(M.IntraMSC_G2U_TOT, 0) AS intra_msc_handover_success_g2u,
    100.0 * M.IntraMSC_U2G_SUCC / NULLIF(M.IntraMSC_U2G_TOT, 0) AS intra_msc_handover_success_u2g,
    
    -- MAP Success Rate
    100.0 * (MAP.NMAP1SUCC + MAP.NMAP2SUCC + MAP.NMAP3SUCC) /
        NULLIF(MAP.NMAP1TOT + MAP.NMAP2TOT + MAP.NMAP3TOT - MAP.NMAP1TCT, 0) AS map_success_rate,
    
    -- CSSR (Complex Calculation)
    CASE 
        WHEN NE.NE_VERSION LIKE 'R1%' THEN
            100 *
            ISNULL(CAST((ISNULL(STS.LOAS_NFTDORG,0)+ISNULL(STS.LOAS_NFTDIEX,0))*1.0 / 
                        NULLIF(ISNULL(STS.LOAS_NOFFORG,0)+ISNULL(STS.LOAS_NOFFIEX,0),0) AS DECIMAL(18,6)), 1) *
            ISNULL(CAST(M.NAUTREQSUCC*1.0 / NULLIF(M.NAUTREQTOT,0) AS DECIMAL(18,6)), 1) *
            ISNULL(CAST((ISNULL(M.NCIPSETSUCC,0)+ISNULL(M.NSECFRRNSCC,0))*1.0 / 
                        NULLIF(ISNULL(M.NCIPATTTOT,0)+ISNULL(M.NSECTORNTOT,0),0) AS DECIMAL(18,6)), 1) *
            ISNULL(CAST(1.0 - ((ISNULL(MT.NUNSUCC,0)+ISNULL(MT.NICONG,0)+ISNULL(MT.NECONG,0))*1.0 / 
                               NULLIF(ISNULL(M.NL3CPROC,0)+ISNULL(TR.NCALLSI,0),0)) AS DECIMAL(18,6)), 1)
        WHEN NE.SOURCE_TYPE='ISBladeHybrid' THEN
            CASE 
                WHEN NE.NE_VERSION='R12A' THEN
                    100*ISNULL(CAST((ISNULL(SPX.NFTORG,0)+ISNULL(SPX.NFTIEX,0))*1.0/ 
                                    NULLIF(ISNULL(SPX.SOROF,0)+ISNULL(SPX.SINOF,0),0) AS DECIMAL(18,6)), 1)
                ELSE
                    100*ISNULL(CAST((ISNULL(SPX.NFTORG,0)+ISNULL(SPX.NFTIEX,0))*1.0/
                                    NULLIF(ISNULL(STS.LOAS_NOFFORG,0)+ISNULL(STS.LOAS_NOFFIEX,0)+ISNULL(SPX.SORRE,0)+ISNULL(SPX.SINRE,0),0) AS DECIMAL(18,6)), 1) *
                    ISNULL(CAST(M.NAUTREQSUCC*1.0/NULLIF(M.NAUTREQTOT,0) AS DECIMAL(18,6)), 1) *
                    ISNULL(CAST((ISNULL(M.NCIPSETSUCC,0)+ISNULL(M.NSECFRRNSCC,0))*1.0/
                                NULLIF(ISNULL(M.NCIPATTTOT,0)+ISNULL(M.NSECTORNTOT,0),0) AS DECIMAL(18,6)), 1) *
                    ISNULL(CAST(1.0-((ISNULL(MT.NUNSUCC,0)+ISNULL(MT.NICONG,0)+ISNULL(MT.NECONG,0))*1.0/
                                     NULLIF(ISNULL(M.NL3CPROC,0)+ISNULL(TR.NCALLSI,0),0)) AS DECIMAL(18,6)), 1)
            END
        ELSE
            100*ISNULL(CAST((ISNULL(STS.LOAS_NFTDORG,0)+ISNULL(STS.LOAS_NFTDIEX,0))*1.0/
                            NULLIF(ISNULL(STS.LOAS_NOFFORG,0)+ISNULL(STS.LOAS_NOFFIEX,0),0) AS DECIMAL(18,6)), 1) *
            ISNULL(CAST(M.NAUTREQSUCC*1.0/NULLIF(M.NAUTREQTOT,0) AS DECIMAL(18,6)), 1) *
            ISNULL(CAST((ISNULL(M.NCIPSETSUCC,0)+ISNULL(M.NSECFRRNSCC,0))*1.0/
                        NULLIF(ISNULL(M.NCIPATTTOT,0)+ISNULL(M.NSECTORNTOT,0),0) AS DECIMAL(18,6)), 1) *
            ISNULL(CAST(1.0-((ISNULL(MT.NUNSUCC,0)+ISNULL(MT.NICONG,0)+ISNULL(MT.NECONG,0))*1.0/
                             NULLIF(ISNULL(M.NL3CPROC,0)+ISNULL(TR.NCALLSI,0),0)) AS DECIMAL(18,6)), 1)
    END AS cssr,

    -- Traffic
    MT.ERLANG,
    100.0 * (1 - (MT.NUNSUCC + MT.NECONG + MT.NICONG) / NULLIF(MT.NCALLS, 0)) AS call_sr,
    
    -- PRBT
    100.0 * M.PRBTSUCC / NULLIF(M.PRBTTOT, 0) AS prbt_invocations_sr,
    
    -- VLR Subscribers
    VLR.VLR_Subs_Registered,
    VLR.VLR_Subs_Active

FROM MSC_AGG M
LEFT JOIN MTRAFTYPE_AGG MT ON M.ELEM = MT.ELEM AND M.DATE_ID = MT.DATE_ID
LEFT JOIN TRUNKROUTE_AGG TR ON M.ELEM = TR.ELEM AND M.DATE_ID = TR.DATE_ID
LEFT JOIN STS_TOT_AGG STS ON M.ELEM = STS.ELEM AND M.DATE_ID = STS.DATE_ID
LEFT JOIN SPXLC_AGG SPX ON M.ELEM = SPX.ELEM AND M.DATE_ID = SPX.DATE_ID
LEFT JOIN SGS_AGG SGS ON M.ELEM = SGS.ELEM AND M.DATE_ID = SGS.DATE_ID
LEFT JOIN MSCOBJ_AGG MSCOBJ ON M.ELEM = MSCOBJ.ELEM AND M.DATE_ID = MSCOBJ.DATE_ID
LEFT JOIN MAP_AGG MAP ON M.ELEM = MAP.ELEM AND M.DATE_ID = MAP.DATE_ID
LEFT JOIN RNC_AGG RNC ON M.ELEM = RNC.ELEM AND M.DATE_ID = RNC.DATE_ID
LEFT JOIN VLR_AGG VLR ON M.ELEM = VLR.ELEM AND M.DATE_ID = VLR.DATE_ID
LEFT JOIN NE_INFO NE ON M.ELEM = NE.NE_NAME

ORDER BY M.DATE_ID DESC, M.ELEM;
